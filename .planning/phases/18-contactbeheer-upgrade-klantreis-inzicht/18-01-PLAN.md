      if (localStorage.getItem(MIGRATION_KEY) === 'true') return;

      const mapping: Record<string, string> = {
        coordinator: 'gebruiker',
        mt: 'beslisser',
        finance: 'inkoper',
      };

      for (const [oldPos, newPos] of Object.entries(mapping)) {
        await supabase.from('contacts')
          .update({ dmu_position: newPos })
          .eq('dmu_position', oldPos);
      }

      localStorage.setItem(MIGRATION_KEY, 'true');
    }

    5. Create a new React Query hook following the exact pattern from useContacts.ts:
    ```typescript
    import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
    import { supabase } from '@/lib/supabase/client';
    import type { SystemEvent } from '@/db/types';
    import { addSystemEvent } from '@/db/operations';

    function mapSystemEventRow(row: Record<string, unknown>): SystemEvent {
      return {
        id: row.id as string,
        schoolId: row.school_id as string,
        timestamp: row.created_at as string,
        eventType: row.event_type as SystemEvent['eventType'],
        description: row.description as string,
        metadata: (row.metadata as Record<string, string>) ?? undefined,
        userId: (row.user_id as string) ?? undefined,
      };
    }

    export function useSystemEvents(schoolId: string) {
      return useQuery({
        queryKey: ['system-events', schoolId],
        queryFn: async () => {
          const { data, error } = await supabase
            .from('system_events')
            .select('*')
            .eq('school_id', schoolId)
            .order('created_at', { ascending: false });
          if (error) throw error;
          return (data ?? []).map(mapSystemEventRow);
        },
        enabled: !!schoolId,
      });
    }

    export function useAddSystemEvent() {
      const queryClient = useQueryClient();
      return useMutation({
        mutationFn: async ({ schoolId, event }: {
          schoolId: string;
          event: Omit<SystemEvent, 'id' | 'timestamp' | 'schoolId'>;
        }) => {
          await addSystemEvent(schoolId, event);
        },
        onSuccess: (_, variables) => {
          queryClient.invalidateQueries({ queryKey: ['system-events', variables.schoolId] });
        },
      });
    }
    ```
