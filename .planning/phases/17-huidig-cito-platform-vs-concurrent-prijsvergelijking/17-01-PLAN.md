    ```typescript
    interface DetectScenarioOptions {

      forRetentionComparison?: boolean;

        hasMigrationModules && !hasCompetitorModules && !hasUpsellModules
          ? (options?.forRetentionComparison ? 'C' : 'B')
          : 'A';

      return { recommended, /* ...rest unchanged... */ };
    }
    ```
