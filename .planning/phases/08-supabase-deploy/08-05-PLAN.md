import Anthropic from '@anthropic-ai/sdk';
import { createClient } from '@supabase/supabase-js';

Export a `POST` handler using Web Standard Request/Response (NOT legacy req/res pattern):
1. Extract Bearer token from Authorization header. If missing: return 401 "Unauthorized".
2. Verify token: `const { data: { user }, error } = await supabaseAdmin.auth.getUser(token)`. If error or no user: return 401.
3. Parse JSON body: `const { notes, systemPrompt } = await request.json()`. Validate notes is a non-empty string. If invalid: return 400 "Notes zijn verplicht".
4. Call Anthropic API:
```typescript
const response = await anthropic.messages.create({
  model: 'claude-haiku-4-5',
  max_tokens: 1024,
  system: systemPrompt || DEFAULT_SYSTEM_PROMPT,
  messages: [{ role: 'user', content: `Analyseer de volgende aantekeningen en extraheer de gestructureerde schoolgegevens:\n\n${notes}` }],
});
```
5. Return JSON response: `new Response(JSON.stringify(response), { headers: { 'Content-Type': 'application/json' } })`
6. Wrap entire handler in try/catch. On error: return 500 "Er is een fout opgetreden bij de AI-verwerking".

Include the DEFAULT_SYSTEM_PROMPT constant (copy from existing ai-intake.ts SYSTEM_PROMPT).

```typescript
export async function extractIntakeFromNotes(notes: string): Promise<IntakeExtraction> {
  const { supabase } = await import('@/lib/supabase/client');
  const { data: { session } } = await supabase.auth.getSession();
  if (!session) throw new Error('Niet ingelogd. Log opnieuw in om AI-functies te gebruiken.');

  const response = await fetch('/api/ai-intake', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'Authorization': `Bearer ${session.access_token}`,
    },
    body: JSON.stringify({ notes, systemPrompt: SYSTEM_PROMPT }),
  });

  if (!response.ok) {
    const text = await response.text();
    throw new Error(text || 'AI-verwerking mislukt. Probeer het opnieuw.');
  }

  const data = await response.json();
  // Parse the Anthropic response content to extract structured data
  const content = data.content[0];
  if (content.type !== 'text') throw new Error('Onverwacht AI-antwoord');

  const parsed = IntakeExtractionSchema.parse(JSON.parse(content.text));
  return parsed;
}
```
